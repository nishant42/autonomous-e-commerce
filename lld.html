Action Proposal contract

// The ONLY way an agent causes a side effect
type ActionProposal = {
  idempotencyKey: string;           // see derivation below
  actionType: "CreatePurchaseOrder" | "AmendPO" | "ChangePrice" | "IssueRefund" | "SetAdBudget" | ...;
  params: Record<string, unknown>;  // validated against the actionType's JSON Schema
  target: { aggregate: "purchase_order" | "sku_price" | ...; id: string; expectedVersion?: number };
  agent: { id: string; version: string; identity: string };   // identity from token, not from payload
  rationale: string;                // human-readable; must cite evidence ids
  evidence: Array<{ kind: "tool_result" | "doc" | "policy"; ref: string }>;
  asOf: { ledgerLsn: string; eventOffset: number };
  correlationId: string; caseId?: string;
};

Action Gateway algorithm

Idempotency key derivation

idempotency_key = sha256(
  action_type
  + business_key        // e.g. "sup_yd:ETOILE-BATH-BLANC:2026-W40" for a reorder cycle
  + intent_version      // bumped only when the agent deliberately makes a *new* decision
)
// Same decision retried (crash, timeout, duplicate event) → same key → one effect.
// A deliberate second PO in the same cycle must bump intent_version, which policy counts and limits.

CREATE UNIQUE INDEX ux_action_idem ON action_proposal (idempotency_key);
-- Execution also passes a vendor idempotency key (e.g. Stripe Idempotency-Key header),
-- so even a gateway bug cannot double-charge.

Policy as code (example)

package gateway.procurement

default tier := "DENY"

po_value := input.params.total_eur
env := data.envelope.procurement

tier := "T0" if { po_value <= env.po_tiers_eur.T0; not anomaly_influenced }
tier := "T1" if { po_value <= env.po_tiers_eur.T1; po_value > env.po_tiers_eur.T0; not anomaly_influenced }
tier := "T2" if { po_value <= env.po_tiers_eur.T2; anomaly_influenced }
tier := "T2" if { po_value <= env.po_tiers_eur.T2; po_value > env.po_tiers_eur.T1 }
tier := "T3" if { po_value > env.po_tiers_eur.T2 }

anomaly_influenced if input.evidence_flags.forecast_anomaly_influenced

deny contains "daily cap exceeded" if {
  data.counters.procurement_today_eur + po_value > env.daily_cap_eur
}
deny contains "cover cap exceeded" if {
  input.derived.weeks_cover_after > env.max_weeks_cover_after_po
}

Supplier adapter interface

interface SupplierChannel {
  readonly kind: "api" | "edi" | "portal" | "email" | "sftp";
  capabilities(): { sendPO: boolean; ack: "sync" | "async" | "none"; asn: boolean; invoice: boolean; stockFeed: boolean };
  sendPurchaseOrder(po: CanonicalPO, key: string): Promise<TransmitReceipt>;   // idempotent on key
  pollInbound(since: Cursor): AsyncIterable<RawInbound>;                     // acks, ASNs, invoices, emails
  normalise(raw: RawInbound): Promise<CanonicalSupplierEvent | NeedsExtraction>;
}
// NeedsExtraction → Document Extractor (maker) → Verifier (checker) → CanonicalSupplierEvent
// Canonical events: supplier.ack · supplier.delay · supplier.price_change · asn.received · invoice.received

Durable workflow (Temporal, abridged)

export async function reorderWorkflow(input: { supplierId: string; skuId: string; cycle: string }) {
  const rec = await activities.planReorder(input);                    // deterministic planner
  if (!rec.needed) return;

  const proposal = await activities.runAgent("procurement-agent", { rec }, { startToCloseTimeout: "5m", retry: { maximumAttempts: 3 } });
  const decision = await activities.submitToGateway(proposal);        // tiering happens here

  if (decision.tier === "T3") {
    const ok = await condition(() => ownerSignal !== undefined, "8h"); // human signal or timeout
    if (!ok) return activities.applySafeDefault(proposal);            // e.g. P50 quantity
  }

  const ack = await Promise.race([
    waitForSignal("supplierAck"),
    sleep("48h").then(() => activities.chaseSupplier(proposal.poId)), // escalating chase ladder
  ]);
  await activities.trackToReceipt(proposal.poId);                     // ASN → 3PL receipt
}
